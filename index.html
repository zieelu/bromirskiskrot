Skrócona instrukcja
1. na rtr i vm01 i vm02 login na root i ip a
2. Na RTR : nano /etc/network/interfaces
dopisujemy to na czerwono

auto lo

iface lo inet loopback



allow-hotplug eth0

auto eth0

iface eth0 inet dhcp



auto eth1

iface eth1 inet static

address 10.41.12.254

netmask 255.255.255.0



auto eth2

iface eth2 inet static

address 192.168.76.12

netmask 255.255.255.0

gateway 192.168.76.1
dns-nameservers 8.8.8.8

3. zapisuje ctrl+s

4. systemctl restart networking






5. VM01 nano /etc/network/interfaces
auto lo
iface lo inet loopback

allow-hotplug eth0
auto eth0
iface eth0 inet dhcp

auto eth1
iface eth1 inet static
address 10.41.12.201
netmask 255.255.255.0
gateway 10.41.12.254
dns-nameservers 8.8.8.8

6. zapis
7. Systemctl restart networking
8. VM02 nano /etc/netplan/50-cloud-init.yaml
network:
version: 2
ethernets:
eth1:
dhcp4: no
addresses:
- 10.41.12.202/24
optional: true
routes:
- to: default
via: 10.41.12.254
nameservers:
addresses:
- 8.8.8.8




9. Netplan apply
10.

NAT + DNAT

RTR wprowadzamy
echo 1 > /proc/sys/net/ipv4/ip_forward
iptables -t nat -F
iptables -F
iptables -t nat -A POSTROUTING -s 10.41.12.0/24 -o eth2 -j MASQUERADE
iptables -A FORWARD -i eth1 -o eth2 -j ACCEPT
iptables -A FORWARD -i eth2 -o eth1 -m state --state ESTABLISHED,RELATED -j ACCEPT
-- sprawdź pingi vm01
ustawiamy DNat
iptables -t nat -A PREROUTING -i eth2 -p tcp --dport 3301 -j DNAT --to 10.41.12.201:22
iptables -t nat -A PREROUTING -i eth2 -p tcp --dport 1102 -j DNAT --to 10.41.12.202:22
iptables -t nat -A PREROUTING -i eth2 -p tcp --dport 5011 -j DNAT --to 10.41.12.201:80
iptables -t nat -A PREROUTING -i eth2 -p tcp --dport 8762 -j DNAT --to 10.41.12.202:80

cat > /etc/dhcp/dhcpd.conf <<'EOF'
default-lease-time 600;
max-lease-time 7200;
authoritative;

subnet 10.41.12.0 netmask 255.255.255.0 {

Nginx - wersja klasyczna

zadanie 2 na Nginx wyglądałoby prawie tak samo, tylko zamiast Apache VirtualHostów robisz server blocki.

VM01

apt update
apt install nginx -y
systemctl enable nginx
systemctl restart nginx

echo "VM01" > /var/www/html/index.html

mkdir -p /var/www/12.net
mkdir -p /var/www/122921.student.san.edu.pl

echo "VM01 - 12.net" > /var/www/12.net/index.html
echo "VM01 - 122921.student.san.edu.pl" > /var/www/122921.student.san.edu.pl/index.html

cat > /etc/nginx/sites-available/12.net <<'EOF'
server {
listen 80;
server_name 12.net;
root /var/www/12.net;
index index.html;
}
EOF

cat > /etc/nginx/sites-available/122921.student.san.edu.pl <<'EOF'
server {
listen 80;
server_name 122921.student.san.edu.pl;
root /var/www/122921.student.san.edu.pl;
index index.html;
}
EOF

ln -s /etc/nginx/sites-available/12.net /etc/nginx/sites-enabled/
ln -s /etc/nginx/sites-available/122921.student.san.edu.pl /etc/nginx/sites-enabled/

nginx -t
systemctl reload nginx

curl -H "Host: 122921.student.san.edu.pl" http://localhost

echo "VM02 - 12.lodz" > /var/www/12.lodz/index.html

cat > /etc/nginx/sites-available/12.lodz <<'EOF'
server {
listen 80;
server_name 12.lodz 12.net.pl;
root /var/www/12.lodz;
index index.html;
}
EOF

ln -s /etc/nginx/sites-available/12.lodz /etc/nginx/sites-enabled/

nginx -t
systemctl reload nginx

Test:
curl http://localhost
curl -H "Host: 12.lodz" http://localhost
curl -H "Host: 12.net.pl" http://localhostRóżnica główna:
